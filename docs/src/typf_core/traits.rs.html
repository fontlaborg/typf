1//! The contracts that bind every backend together.
2//!
3//! Typf's pipeline has three interchangeable steps, each defined by a trait:
4//!
5//! ```text
6//! "Hello, مرحبا"  ──[Shaper]──▶  positioned glyphs  ──[Renderer]──▶  pixels/paths  ──[Exporter]──▶  PNG/SVG/…
7//! ```
8//!
9//! Swap any step without touching the others. Five shaper implementations ×
10//! seven renderer implementations = 35 valid backend combinations, all wired
11//! through these four traits.
12//!
13//! ## The Players
14//!
15//! - [`Stage`] — The foundation every pipeline component builds upon
16//! - [`FontRef`] — Your window into font data and metrics
17//! - [`Shaper`] — Converts Unicode text into positioned glyphs (step 1)
18//! - [`Renderer`] — Draws those glyphs into pixels or vector paths (step 2)
19//! - [`Exporter`] — Encodes the rendered output into a file format (step 3)
20//!
21//! ## What "shaping" means
22//!
23//! Text is not a sequence of pictures. "A" in isolation looks different from
24//! "A" kerned next to "V". Arabic letters change shape depending on whether
25//! they start, continue, or end a word. Hindi consonants stack vertically into
26//! ligatures. Even "fi" in a well-made Latin font is a single merged glyph.
27//!
28//! A [`Shaper`] reads the font's OpenType rules (GSUB/GPOS tables) and figures
29//! out *which* glyph IDs to use, in *what order*, at *what positions*. The
30//! result is a [`ShapingResult`]: a flat list of
31//! glyph IDs each carrying an (x, y) offset from the run origin.
32//!
33//! ## What "rendering" means
34//!
35//! Once you know which glyphs go where, you need to draw them. Each glyph in
36//! a font is stored as a set of Bézier curves (the outline). A [`Renderer`]
37//! reads those curves and either rasterizes them into a pixel grid (bitmap
38//! output) or traces them as SVG paths (vector output).
39//!
40//! ## What "exporting" means
41//!
42//! The rendered pixels or paths live in memory as a [`RenderOutput`].
43//! An [`Exporter`] encodes that into bytes you can write to disk — PNG, PNM,
44//! SVG, JSON, etc.

51/// Every pipeline dancer learns these same steps
52///
53/// Implement Stage and your component can join the six-stage procession
54/// that transforms text into rendered output.
55///
56/// ```ignore
57/// struct MyStage;
58///
59/// impl Stage for MyStage {
60///     fn name(&self) -> &'static str {
61///         "my-stage"
62///     }
63///
64///     fn process(&self, context: PipelineContext) -> Result<PipelineContext> {
65///         // Transform the context, pass it forward
66///         Ok(context)
67///     }
68/// }
69/// ```
70pub trait Stage: Send + Sync {
71    /// Who are you? Used for debugging and logging
72    fn name(&self) -> &'static str;
73
74    /// Do your work and pass the context forward
75    ///
76    /// Take the context, make your changes, and return it for the next stage.
77    fn process(&self, context: PipelineContext) -> Result<PipelineContext>;
78}

80/// A uniform window into font data, regardless of the underlying font library.
81///
82/// Shapers and renderers do not care whether the font was loaded by `fontdb`,
83/// `skrifa`, `ttf-parser`, or a custom loader. They only care about this
84/// abstraction: give me the raw bytes, tell me the coordinate scale, look up a
85/// glyph ID by character, and tell me how wide that glyph is.
86///
87/// ## Font coordinate units
88///
89/// Every measurement inside a font file is in "font units" — an arbitrary
90/// integer coordinate space whose scale is set by `units_per_em`. A glyph
91/// that is 700 font-units wide in a font with `units_per_em = 1000` is 70% of
92/// one em wide. At `font_size = 48px`, that glyph is `48 × 0.70 = 33.6px`.
93///
94/// Type 1 and most OTF (CFF) fonts use 1000 units/em.
95/// TrueType (glyf) fonts typically use 2048 units/em.
96///
97/// ```ignore
98/// struct MyFont {
99///     data: Vec<u8>,
100///     // ... your internal state
101/// }
102///
103/// impl FontRef for MyFont {
104///     fn data(&self) -> &[u8] {
105///         &self.data
106///     }
107///
108///     fn units_per_em(&self) -> u16 {
109///         1000
110///     }
111///
112///     fn glyph_id(&self, ch: char) -> Option<GlyphId> {
113///         // Look up the Unicode character in the font's cmap table.
114///         Some(42)
115///     }
116///
117///     fn advance_width(&self, glyph_id: GlyphId) -> f32 {
118///         // Read the hmtx table for this glyph's horizontal advance.
119///         500.0
120///     }
121/// }
122/// ```
123pub trait FontRef: Send + Sync {
124    /// The raw font file bytes (TTF/OTF/WOFF data).
125    ///
126    /// Shapers pass this directly to their underlying library (HarfBuzz, skrifa,
127    /// etc.). Keep the slice alive as long as the font is in use.
128    fn data(&self) -> &[u8];
129
130    /// Shared font bytes for zero-copy downstream consumption.
131    ///
132    /// Implementations that store their bytes in an `Arc` SHOULD override this
133    /// to avoid per-call allocations in downstream libraries that require shared
134    /// ownership (e.g. Vello's `FontData`). Default returns `None`.
135    fn data_shared(&self) -> Option<Arc<dyn AsRef<[u8]> + Send + Sync>> {
136        None
137    }
138
139    /// The font's internal coordinate scale (units per em).
140    ///
141    /// Divide any font-unit measurement by this value to get a fraction of one
142    /// em, then multiply by the target `font_size` in pixels to get pixels.
143    /// Common values: 1000 (CFF/Type 1), 2048 (TrueType).
144    fn units_per_em(&self) -> u16;
145
146    /// Font-wide metrics in font units: ascent, descent, and line gap.
147    ///
148    /// Backends that parse OpenType tables SHOULD provide this so callers can
149    /// compute baselines without depending on a specific font parser. Returns
150    /// `None` if the implementation cannot or does not parse these tables.
151    fn metrics(&self) -> Option<FontMetrics> {
152        None
153    }
154
155    /// Map a Unicode character to the font-specific glyph ID that represents it.
156    ///
157    /// This is a cmap lookup. Returns `None` when the font does not contain
158    /// a glyph for `ch` (the font is missing that character). Note that shapers
159    /// may return *different* glyph IDs for the same character after applying
160    /// OpenType substitution — this method only reflects the raw cmap mapping.
161    fn glyph_id(&self, ch: char) -> Option<GlyphId>;
162
163    /// Horizontal advance of `glyph_id` in font units.
164    ///
165    /// This is the cursor-advance distance read from the `hmtx` table: how far
166    /// right the pen moves after drawing this glyph. Multiply by
167    /// `(font_size / units_per_em)` to convert to pixels.
168    fn advance_width(&self, glyph_id: GlyphId) -> f32;
169
170    /// Total number of glyphs in the font (from the `maxp` table).
171    ///
172    /// Used to validate shaper output — a glyph ID ≥ `glyph_count` is invalid.
173    /// Returns `None` if the implementation cannot provide this cheaply.
174    fn glyph_count(&self) -> Option<u32> {
175        None
176    }
177
178    /// Variable font axes from the `fvar` table.
179    ///
180    /// A variable font encodes a continuous design space along named axes such
181    /// as `wght` (weight), `wdth` (width), `slnt` (slant), or `ital` (italic).
182    /// Each axis has a min, default, and max value; callers pass specific
183    /// coordinates via [`ShapingParams::variations`][crate::ShapingParams] and
184    /// [`RenderParams::variations`][crate::RenderParams].
185    ///
186    /// Returns `None` for static (non-variable) fonts or if the font file
187    /// cannot be parsed. Returns `Some(empty vec)` if the `fvar` table exists
188    /// but declares no axes.
189    fn variation_axes(&self) -> Option<Vec<VariationAxis>> {
190        None
191    }
192
193    /// Returns `true` if this font has at least one variable axis.
194    fn is_variable(&self) -> bool {
195        self.variation_axes().is_some_and(|axes| !axes.is_empty())
196    }
197}

199/// Step 1 of the pipeline: turn Unicode text into positioned glyphs.
200///
201/// A shaper reads the font's OpenType layout tables and answers: *which glyph
202/// IDs should appear, in what order, at what (x, y) offsets?*
203///
204/// That question is trivial for ASCII — one codepoint, one glyph, advance by
205/// the glyph width. It is not trivial for:
206///
207/// - **Arabic/Hebrew**: runs right-to-left; letters have up to four contextual
208///   forms (isolated, initial, medial, final) selected from the GSUB table.
209/// - **Devanagari/Bengali**: consonant clusters combine into conjuncts; vowel
210///   signs reorder visually relative to their base consonant.
211/// - **Thai/Khmer**: no spaces between words; vowels and tone marks stack
212///   above or below their base letter.
213/// - **Latin**: ligatures (`fi` → one glyph), kerning (`AV` tighter than `AX`),
214///   small-caps, old-style figures — all driven by GSUB/GPOS feature tags.
215///
216/// ## Available implementations
217///
218/// | Crate | Name | What it uses |
219/// |---|---|---|
220/// | `typf-shape-none` | `"none"` | Simple LTR passthrough — no OpenType features |
221/// | `typf-shape-hb` | `"harfbuzz"` | HarfBuzz C library via FFI — industry standard |
222/// | `typf-shape-hr` | `"harfrust"` | Pure Rust HarfBuzz port — zero C deps |
223/// | `typf-shape-icu-hb` | `"icu-hb"` | ICU normalization + HarfBuzz — full Unicode pipeline |
224/// | `typf-shape-ct` | `"coretext"` | macOS CoreText — native Apple typography |
225pub trait Shaper: Send + Sync {
226    /// Identify yourself in logs and error messages.
227    fn name(&self) -> &'static str;
228
229    /// Turn `text` into a list of positioned glyphs using `font` and `params`.
230    ///
231    /// The returned [`ShapingResult`] contains one [`PositionedGlyph`]
232    /// per rendered glyph (which may differ in count from the input codepoints —
233    /// ligatures merge multiple codepoints; some scripts expand one codepoint
234    /// into several glyphs).
235    fn shape(
236        &self,
237        text: &str,
238        font: Arc<dyn FontRef>,
239        params: &ShapingParams,
240    ) -> Result<ShapingResult>;
241
242    /// Returns `true` if this shaper can correctly handle the named script.
243    ///
244    /// Script names follow ISO 15924 four-letter codes: `"Arab"`, `"Deva"`,
245    /// `"Latn"`, `"Thai"`, etc. Returns `false` by default; implementations
246    /// should explicitly declare what they support.
247    fn supports_script(&self, _script: &str) -> bool {
248        false
249    }
250
251    /// Flush any internally cached shaping results.
252    fn clear_cache(&self) {}
253}

255/// Step 2 of the pipeline: draw positioned glyphs into pixels or vector paths.
256///
257/// A renderer receives a [`ShapingResult`] (the list of positioned glyph IDs)
258/// and a font, then reads each glyph's outline from the font file (Bézier
259/// curves stored in the `glyf`, `CFF`, or `CFF2` table), and either:
260///
261/// - **Rasterizes** the curves into a pixel grid with anti-aliasing, or
262/// - **Traces** the curves into SVG path data for vector output.
263///
264/// The output is a [`RenderOutput`] — either a [`BitmapData`]
265/// (RGBA pixels) or [`VectorData`] (SVG/PDF text).
266///
267/// ## Available implementations
268///
269/// | Crate | Name | Notes |
270/// |---|---|---|
271/// | `typf-render-opixa` | `"opixa"` | Pure Rust, SIMD-optimized, monochrome only |
272/// | `typf-render-skia` | `"skia"` | tiny-skia (Rust port of Skia), full color font support |
273/// | `typf-render-zeno` | `"zeno"` | Pure Rust zeno rasterizer, 256-level AA, color fonts |
274/// | `typf-render-vello-cpu` | `"vello-cpu"` | Vello CPU path, pure Rust, COLR/bitmap color fonts |
275/// | `typf-render-vello` | `"vello"` | Vello GPU (Metal/Vulkan/DX12), outline-only today |
276/// | `typf-render-cg` | `"coregraphics"` | macOS CoreGraphics — native Apple rendering |
277/// | `typf-render-json` | `"json"` | Emits HarfBuzz-compatible JSON instead of pixels |
278pub trait Renderer: Send + Sync {
279    /// Identify yourself in logs and error messages.
280    fn name(&self) -> &'static str;
281
282    /// Draw `shaped` glyphs using `font`'s outlines and return the visual output.
283    ///
284    /// Implementations must respect `params.output` (bitmap vs vector) and
285    /// `params.glyph_sources` (which glyph table to prefer for color fonts).
286    fn render(
287        &self,
288        shaped: &ShapingResult,
289        font: Arc<dyn FontRef>,
290        params: &RenderParams,
291    ) -> Result<RenderOutput>;
292
293    /// Returns `true` if this renderer can produce the named output format.
294    ///
295    /// Format strings match file extensions: `"png"`, `"svg"`, `"json"`.
296    /// Returns `false` by default; implementations should declare support.
297    fn supports_format(&self, _format: &str) -> bool {
298        false
299    }
300
301    /// Free up any internally cached glyph bitmaps or path data.
302    fn clear_cache(&self) {}
303}

305/// Step 3 of the pipeline: encode rendered output as bytes ready to write to disk.
306///
307/// An exporter takes a [`RenderOutput`] (pixels or vector paths already in
308/// memory) and serializes it into the final file format — PNG, PNM, SVG, JSON,
309/// etc. The returned `Vec<u8>` can be written directly to a file or streamed
310/// over a network.
311pub trait Exporter: Send + Sync {
312    /// Identify yourself in logs and error messages.
313    fn name(&self) -> &'static str;
314
315    /// Encode `output` and return the file bytes.
316    fn export(&self, output: &RenderOutput) -> Result<Vec<u8>>;
317
318    /// File extension for this format, without the leading dot (e.g. `"png"`).
319    fn extension(&self) -> &'static str;
320
321    /// MIME type for HTTP Content-Type headers (e.g. `"image/png"`).
322    fn mime_type(&self) -> &'static str;
323}
