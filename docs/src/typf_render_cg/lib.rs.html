30/// Bridge between our font bytes and CoreGraphics' data expectations
31///
32/// CoreGraphics needs data that lives as long as the font object does.
33/// This wrapper keeps our font data alive with proper reference counting.
34struct ProviderData {
35    bytes: Arc<[u8]>,
36}
37
38/// Direct access to macOS's professional text rendering pipeline
39///
40/// This isn't just another renderer—it's a first-class citizen on macOS,
41/// using the exact same APIs that Safari, Pages, and the system text renderer
42/// use. When you need pixel-perfect consistency with the platform, this is it.
43pub struct CoreGraphicsRenderer;

51    /// Turns raw font bytes into a CoreGraphics-ready font object
52    ///
53    /// This is where we bridge Typf's font loading with CoreGraphics' expectations.
54    /// The data provider pattern ensures the font data stays alive as long as
55    /// CoreGraphics needs it.
56    fn create_cg_font(data: &[u8]) -> Result<CFRetained<CGFont>> {
57        // Wrap our font data in an Arc for proper lifetime management
58        let provider_data = Arc::new(ProviderData {
59            bytes: Arc::from(data),
60        });
61
62        // Create data provider from bytes
63        let data_ptr = provider_data.bytes.as_ptr();
64        let data_len = provider_data.bytes.len();
65
66        // Use CGDataProvider::with_data with a release callback that drops our Arc
67        let provider = unsafe {
68            let boxed = Box::into_raw(Box::new(provider_data));
69            extern "C-unwind" fn release_callback(
70                info: *mut c_void,
71                _data: NonNull<c_void>,
72                _size: usize,
73            ) {
74                unsafe {
75                    let _ = Box::from_raw(info as *mut Arc<ProviderData>);
76                }
77            }
78            CGDataProvider::with_data(
79                boxed as *mut c_void,
80                data_ptr as *const c_void,
81                data_len,
82                Some(release_callback),
83            )
84        };
85
86        let provider = provider.ok_or_else(|| {
87            TypfError::RenderingFailed(RenderError::BackendError(
88                "Failed to create CGDataProvider".to_string(),
89            ))
90        })?;
91
92        // Let CoreGraphics parse and validate the font data
93        let cg_font = CGFont::with_data_provider(&provider);
94
95        cg_font.ok_or_else(|| {
96            TypfError::RenderingFailed(RenderError::BackendError(
97                "CoreGraphics rejected our font data".to_string(),
98            ))
99        })
100    }

102    /// Figures out how much canvas space our shaped text needs
103    ///
104    /// CoreGraphics needs explicit dimensions, so we calculate the bounding box
105    /// that contains all our positioned glyphs plus any requested padding.
106    /// Returns None if dimensions cannot be computed (NaN/Inf in glyph data).
107    fn calculate_dimensions(shaped: &ShapingResult, params: &RenderParams) -> Option<(u32, u32)> {
108        // Track the extremes of our glyph layout
109        let mut min_x = f32::MAX;
110        let mut max_x = f32::MIN;
111        let mut min_y = f32::MAX;
112        let mut max_y = f32::MIN;
113
114        for glyph in &shaped.glyphs {
115            // Skip glyphs with invalid positions (NaN/Inf from faulty font data)
116            if !glyph.x.is_finite() || !glyph.advance.is_finite() || !glyph.y.is_finite() {
117                continue;
118            }
119            min_x = min_x.min(glyph.x);
120            max_x = max_x.max(glyph.x + glyph.advance);
121            // Estimate vertical bounds using font proportions (80% ascent, 20% descent)
122            min_y = min_y.min(glyph.y - shaped.advance_height * 0.8);
123            max_y = max_y.max(glyph.y + shaped.advance_height * 0.2);
124        }
125
126        // Don't crash on empty text or all-invalid glyphs—give it a minimal reasonable size
127        if shaped.glyphs.is_empty() || min_x == f32::MAX {
128            min_x = 0.0;
129            max_x = 1.0;
130            min_y = 0.0;
131            max_y = 1.0;
132        }
133
134        let content_width = (max_x - min_x).max(1.0);
135        let content_height = (max_y - min_y).max(1.0);
136
137        // Validate computed dimensions are finite
138        if !content_width.is_finite() || !content_height.is_finite() {
139            return None;
140        }
141
142        let padding = params.padding as f32;
143        let width = (content_width + padding * 2.0).ceil().max(1.0) as u32;
144        let height = (content_height + padding * 2.0).ceil().max(1.0) as u32;
145
146        // Final sanity check: dimensions must be positive and within limits
147        let max_width = typf_core::get_max_bitmap_width();
148        let max_height = typf_core::get_max_bitmap_height();
149        let max_pixels = typf_core::get_max_bitmap_pixels();
150        if width == 0 || height == 0 || width > max_width || height > max_height {
151            return None;
152        }
153        if (width as u64 * height as u64) > max_pixels {
154            return None;
155        }
156
157        Some((width, height))
158    }

160    /// Convert Typf's Color type to CoreGraphics' normalized float format
161    fn color_to_rgb(color: &Color) -> (f64, f64, f64, f64) {
162        (
163            color.r as f64 / 255.0,
164            color.g as f64 / 255.0,
165            color.b as f64 / 255.0,
166            color.a as f64 / 255.0,
167        )
168    }

170    /// Convert a 4-character axis tag to its numeric identifier.
171    /// Example: "wght" -> 2003265652 (0x77676874)
172    fn tag_to_axis_id(tag: &str) -> Option<i64> {
173        if tag.len() != 4 {
174            return None;
175        }
176        let bytes = tag.as_bytes();
177        Some(
178            ((bytes[0] as i64) << 24)
179                | ((bytes[1] as i64) << 16)
180                | ((bytes[2] as i64) << 8)
181                | (bytes[3] as i64),
182        )
183    }

185    /// Creates a CTFont from font data with optional variation coordinates applied.
186    ///
187    /// For variable fonts, this applies the specified axis values to produce
188    /// the correct glyph outlines. Without this, variable fonts would always
189    /// render at their default axis values regardless of shaping parameters.
190    fn create_ct_font_with_variations(
191        data: &[u8],
192        font_size: f64,
193        variations: &[(String, f32)],
194    ) -> Result<CFRetained<CTFont>> {
195        // Create CGFont from raw data - this keeps our loaded font data
196        let cg_font = Self::create_cg_font(data)?;
197
198        // Apply variation coordinates if specified
199        if !variations.is_empty() {
200            log::debug!(
201                "CoreGraphicsRenderer: Applying {} variation coordinates",
202                variations.len()
203            );
204
205            // CoreText's variation dictionary requires:
206            // - Keys: CFNumber representing axis identifier (32-bit integer from 4-char tag)
207            // - Values: CFNumber representing axis value
208            let var_pairs: Vec<(CFRetained<CFNumber>, CFRetained<CFNumber>)> = variations
209                .iter()
210                .filter_map(|(tag, value)| {
211                    Self::tag_to_axis_id(tag).map(|axis_id| {
212                        log::debug!(
213                            "CoreGraphicsRenderer: axis '{}' -> id {} = {}",
214                            tag,
215                            axis_id,
216                            value
217                        );
218                        let key = CFNumber::new_i64(axis_id);
219                        let val = CFNumber::new_f64(*value as f64);
220                        (key, val)
221                    })
222                })
223                .collect();
224
225            if !var_pairs.is_empty() {
226                // Create variation dictionary
227                let keys: Vec<&CFNumber> = var_pairs.iter().map(|(k, _)| k.as_ref()).collect();
228                let values: Vec<&CFNumber> = var_pairs.iter().map(|(_, v)| v.as_ref()).collect();
229
230                let var_dict: CFRetained<CFDictionary<CFNumber, CFNumber>> =
231                    CFDictionary::from_slices(&keys, &values);
232
233                // Create font descriptor with variations attribute
234                // kCTFontVariationAttribute is a *const CFString, we need to retain it
235                let var_key_ptr: *const CFString = unsafe { kCTFontVariationAttribute };
236                let var_key = unsafe {
237                    CFRetained::retain(NonNull::new(var_key_ptr as *mut CFString).unwrap())
238                };
239                let keys_for_attrs: Vec<&CFString> = vec![&var_key];
240                // Cast var_dict to untyped CFDictionary for use as attribute value
241                let var_dict_untyped: CFRetained<CFDictionary> =
242                    unsafe { CFRetained::cast_unchecked(var_dict) };
243                let values_for_attrs: Vec<&CFDictionary> = vec![&var_dict_untyped];
244
245                let attrs: CFRetained<CFDictionary<CFString, CFDictionary>> =
246                    CFDictionary::from_slices(&keys_for_attrs, &values_for_attrs);
247
248                // Cast to untyped dictionary for CTFontDescriptor
249                let attrs_untyped: CFRetained<CFDictionary> =
250                    unsafe { CFRetained::cast_unchecked(attrs) };
251
252                let desc = unsafe { CTFontDescriptor::with_attributes(&attrs_untyped) };
253
254                let ct_font = unsafe {
255                    CTFont::with_graphics_font(
256                        &cg_font,
257                        font_size,
258                        ptr::null::<CGAffineTransform>(),
259                        Some(&desc),
260                    )
261                };
262
263                return Ok(ct_font);
264            }
265        }
266
267        // No variations - create CTFont directly from CGFont
268        let ct_font = unsafe {
269            CTFont::with_graphics_font(&cg_font, font_size, ptr::null::<CGAffineTransform>(), None)
270        };
271
272        Ok(ct_font)
273    }

282impl Renderer for CoreGraphicsRenderer {
283    fn name(&self) -> &'static str {
284        "coregraphics"
285    }
286
287    fn render(
288        &self,
289        shaped: &ShapingResult,
290        font: Arc<dyn FontRef>,
291        params: &RenderParams,
292    ) -> Result<RenderOutput> {
293        log::debug!(
294            "CoreGraphicsRenderer: Rendering {} glyphs",
295            shaped.glyphs.len()
296        );
297
298        // Handle empty glyph case
299        if shaped.glyphs.is_empty() {
300            log::debug!("CoreGraphicsRenderer: No glyphs to render");
301            return Ok(RenderOutput::Bitmap(BitmapData {
302                width: 1,
303                height: 1,
304                format: BitmapFormat::Rgba8,
305                data: vec![0, 0, 0, 255], // Transparent pixel
306            }));
307        }
308
309        // Calculate dimensions - bail gracefully if font data is corrupt
310        let (width, height) = Self::calculate_dimensions(shaped, params).ok_or_else(|| {
311            TypfError::RenderingFailed(RenderError::BackendError(
312                "Cannot compute valid bitmap dimensions (font may have corrupt glyph data)"
313                    .to_string(),
314            ))
315        })?;
316
317        log::debug!("CoreGraphicsRenderer: Canvas size {}x{}", width, height);
318
319        // Create bitmap buffer (RGBA, premultiplied alpha)
320        let bytes_per_row = width as usize * 4;
321        let mut buffer = vec![0u8; height as usize * bytes_per_row];
322
323        // Create CGContext
324        let color_space = CGColorSpace::new_device_rgb();
325        let color_space = color_space.ok_or_else(|| {
326            TypfError::RenderingFailed(RenderError::BackendError(
327                "Failed to create CGColorSpace".to_string(),
328            ))
329        })?;
330
331        // kCGImageAlphaPremultipliedLast = 1
332        let bitmap_info = 1u32;
333        let context = unsafe {
334            CGBitmapContextCreate(
335                buffer.as_mut_ptr() as *mut c_void,
336                width as usize,
337                height as usize,
338                8, // bits per component
339                bytes_per_row,
340                Some(&color_space),
341                bitmap_info,
342            )
343        };
344
345        let context = context.ok_or_else(|| {
346            TypfError::RenderingFailed(RenderError::BackendError(
347                "Failed to create CGContext".to_string(),
348            ))
349        })?;
350
351        // Configure antialiasing
352        CGContext::set_should_antialias(Some(&context), params.antialias);
353        CGContext::set_should_smooth_fonts(Some(&context), params.antialias);
354
355        // Fill background (default to transparent if not specified)
356        if let Some(bg_color) = &params.background {
357            let (r, g, b, a) = Self::color_to_rgb(bg_color);
358            CGContext::set_rgb_fill_color(Some(&context), r, g, b, a);
359            CGContext::fill_rect(
360                Some(&context),
361                CGRect {
362                    origin: CGPoint { x: 0.0, y: 0.0 },
363                    size: CGSize {
364                        width: width as f64,
365                        height: height as f64,
366                    },
367                },
368            );
369        } else {
370            // Clear to transparent (RGBA all zeros)
371            CGContext::clear_rect(
372                Some(&context),
373                CGRect {
374                    origin: CGPoint { x: 0.0, y: 0.0 },
375                    size: CGSize {
376                        width: width as f64,
377                        height: height as f64,
378                    },
379                },
380            );
381        }
382
383        // Set text color
384        let (r, g, b, a) = Self::color_to_rgb(&params.foreground);
385        CGContext::set_rgb_fill_color(Some(&context), r, g, b, a);
386
387        // Validate font size before creating CTFont
388        // CoreText requires a positive, finite font size
389        let font_size = shaped.advance_height as f64;
390        if !font_size.is_finite() || font_size <= 0.0 {
391            return Err(TypfError::RenderingFailed(RenderError::BackendError(
392                format!(
393                    "Invalid font size: {}. Font size must be positive and finite.",
394                    font_size
395                ),
396            )));
397        }
398
399        // Create CTFont with variation coordinates applied (critical for variable fonts!)
400        let ct_font =
401            Self::create_ct_font_with_variations(font.data(), font_size, &params.variations)?;
402
403        // Prepare glyph data
404        let glyph_ids: Vec<CGGlyph> = shaped
405            .glyphs
406            .iter()
407            .map(|g| g.id.min(u16::MAX as u32) as CGGlyph)
408            .collect();
409
410        log::debug!(
411            "CoreGraphicsRenderer: Rendering {} glyphs, font_size={}",
412            glyph_ids.len(),
413            shaped.advance_height
414        );
415        if !glyph_ids.is_empty() {
416            log::debug!(
417                "CoreGraphicsRenderer: First glyph: id={}, x={}, y={}",
418                glyph_ids[0],
419                shaped.glyphs[0].x,
420                shaped.glyphs[0].y
421            );
422        }
423
424        // Calculate glyph positions relative to origin (after translate)
425        // CoreGraphics uses bottom-left origin. Calculate baseline position.
426        // Use 0.75 ratio: baseline at 75% from top = 25% from bottom
427        const BASELINE_RATIO: f64 = 0.75;
428        let baseline_y = (height as f64) * (1.0 - BASELINE_RATIO);
429
430        let glyph_positions: Vec<CGPoint> = shaped
431            .glyphs
432            .iter()
433            .map(|g| CGPoint {
434                x: g.x as f64,
435                y: g.y as f64,
436            })
437            .collect();
438
439        if !glyph_positions.is_empty() {
440            log::debug!(
441                "CoreGraphicsRenderer: First glyph position: x={}, y={}, baseline_y={}",
442                glyph_positions[0].x,
443                glyph_positions[0].y,
444                baseline_y
445            );
446        }
447
448        // Render glyphs using CTFont
449        CGContext::save_g_state(Some(&context));
450        CGContext::translate_ctm(Some(&context), params.padding as f64, baseline_y);
451        CGContext::set_text_drawing_mode(Some(&context), CGTextDrawingMode::Fill);
452
453        // Draw glyphs using CTFont::draw_glyphs
454        unsafe {
455            ct_font.draw_glyphs(
456                NonNull::new_unchecked(glyph_ids.as_ptr() as *mut CGGlyph),
457                NonNull::new_unchecked(glyph_positions.as_ptr() as *mut CGPoint),
458                glyph_ids.len(),
459                &context,
460            );
461        }
462
463        CGContext::restore_g_state(Some(&context));
464
465        // Return bitmap data
466        Ok(RenderOutput::Bitmap(BitmapData {
467            width,
468            height,
469            format: BitmapFormat::Rgba8,
470            data: buffer,
471        }))
472    }
473
474    fn supports_format(&self, format: &str) -> bool {
475        matches!(format, "bitmap" | "rgba")
476    }
477}
