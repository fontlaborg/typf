1//! SVG export support for Typf
2//!
3//! This module provides SVG vector output from shaped text.
4//!
5//! ## Features
6//!
7//! - Direct glyph outline export to SVG paths
8//! - Proper coordinate transformation and scaling
9//! - Support for foreground colors and opacity
10//! - Compact SVG output with optimized path commands
11//! - Viewbox calculation for proper sizing
12//! - Optional bitmap glyph embedding as base64 PNG (with `bitmap-embed` feature)
13//!
14//! ## Feature Flags
15//!
16//! - `bitmap-embed` - Enables embedding bitmap glyphs (CBDT/sbix) as base64 PNG `<image>` elements
17//!
18//! Community project by FontLab - <https://www.fontlab.org/>
19

34/// SVG exporter for vector text output
35#[derive(Debug)]
36pub struct SvgExporter {

44impl SvgExporter {

60    /// Enable or disable bitmap glyph embedding (requires `bitmap-embed` feature)
61    ///
62    /// When enabled, bitmap-only glyphs (CBDT/sbix) are embedded as base64 PNG `<image>` elements.
63    /// When disabled (or when the feature is not compiled), bitmap-only glyphs are emitted as a
64    /// placeholder box so they do not silently disappear from the output.
65    #[cfg(feature = "bitmap-embed")]
66    pub fn with_bitmap_embedding(mut self, embed: bool) -> Self {
67        self.embed_bitmaps = embed;
68        self
69    }

71    /// Export shaped text to SVG
72    ///
73    /// # Arguments
74    ///
75    /// * `shaped` - Shaping result containing glyph positions
76    /// * `font` - Font reference for glyph outline extraction
77    /// * `foreground` - Text color
78    ///
79    /// # Returns
80    ///
81    /// Complete SVG document as a string
82    pub fn export(
83        &self,
84        shaped: &ShapingResult,
85        font: Arc<dyn FontRef>,
86        foreground: Color,
87    ) -> Result<String> {
88        // Calculate viewBox dimensions
89        let width = shaped.advance_width + self.padding * 2.0;
90        let height = shaped.advance_height + self.padding * 2.0;
91
92        let mut svg = String::new();
93
94        // SVG header
95        writeln!(&mut svg, r#"<?xml version="1.0" encoding="UTF-8"?>"#)
96            .map_err(|e| ExportError::WriteFailed(e.to_string()))?;
97
98        writeln!(
99            &mut svg,
100            r#"<svg xmlns="http://www.w3.org/2000/svg" viewBox="0 0 {:.2} {:.2}" width="{:.0}" height="{:.0}">"#,
101            width, height, width, height
102        )
103        .map_err(|e| ExportError::WriteFailed(e.to_string()))?;
104
105        // Background (optional - only if not fully transparent)
106        // Skip for now to keep output minimal
107
108        // Extract and render each glyph
109        let baseline_y = height * 0.8; // Position baseline at 80% of height
110        let scale = shaped.advance_height / font.units_per_em() as f32;
111
112        for glyph in &shaped.glyphs {
113            // Calculate glyph position
114            let x = self.padding + glyph.x;
115            let y = baseline_y + glyph.y;
116
117            // Try to extract outline path
118            let outline_result = self.extract_glyph_outline(&font, glyph.id, scale);
119
120            match outline_result {
121                Ok(GlyphOutline::Path(path)) => {
122                    // Write path element with transform
123                    writeln!(
124                        &mut svg,
125                        r#"  <path d="{}" fill="rgb({},{},{})" fill-opacity="{:.2}" transform="translate({:.2},{:.2})"/>"#,
126                        path,
127                        foreground.r,
128                        foreground.g,
129                        foreground.b,
130                        foreground.a as f32 / 255.0,
131                        x,
132                        y
133                    )
134                    .map_err(|e| ExportError::WriteFailed(e.to_string()))?;
135                },
136                Ok(GlyphOutline::EmptyOutline) => {
137                    // Intentionally empty glyph (e.g. space). Emit nothing.
138                },
139                Ok(GlyphOutline::BitmapOnly) => {
140                    #[cfg(feature = "bitmap-embed")]
141                    let mut wrote_bitmap = false;
142                    #[cfg(not(feature = "bitmap-embed"))]
143                    let wrote_bitmap = false;
144                    #[cfg(feature = "bitmap-embed")]
145                    {
146                        if self.embed_bitmaps {
147                            if let Some(image_element) = self.try_embed_bitmap_glyph(
148                                &font,
149                                glyph.id,
150                                shaped.advance_height,
151                                x,
152                                y,
153                            )? {
154                                writeln!(&mut svg, "{}", image_element)
155                                    .map_err(|e| ExportError::WriteFailed(e.to_string()))?;
156                                wrote_bitmap = true;
157                            }
158                        }
159                    }
160
161                    if !wrote_bitmap {
162                        self.write_missing_glyph_placeholder(
163                            &mut svg,
164                            x,
165                            y,
166                            glyph.advance,
167                            shaped.advance_height,
168                            foreground,
169                        )?;
170                    }
171                },
172                Err(_) => {
173                    // Glyph extraction failed - try bitmap as fallback
174                    #[cfg(feature = "bitmap-embed")]
175                    if self.embed_bitmaps {
176                        if let Some(image_element) = self.try_embed_bitmap_glyph(
177                            &font,
178                            glyph.id,
179                            shaped.advance_height,
180                            x,
181                            y,
182                        )? {
183                            writeln!(&mut svg, "{}", image_element)
184                                .map_err(|e| ExportError::WriteFailed(e.to_string()))?;
185                            continue;
186                        }
187                    }
188                    // No bitmap fallback available - skip silently
189                },
190            }
191        }
192
193        // SVG footer
194        writeln!(&mut svg, "</svg>").map_err(|e| ExportError::WriteFailed(e.to_string()))?;
195
196        Ok(svg)
197    }

199    /// Extract glyph outline as SVG path string
200    ///
201    /// The SVG exporter treats glyphs with bitmap strikes but no outline as `BitmapOnly`, so callers
202    /// can decide whether to embed a bitmap image or emit a placeholder.
203    fn extract_glyph_outline(
204        &self,
205        font: &Arc<dyn FontRef>,
206        glyph_id: u32,
207        scale: f32,
208    ) -> Result<GlyphOutline> {
209        let font_data = font.data();
210        let font_ref = skrifa::FontRef::new(font_data)
211            .map_err(|_| ExportError::EncodingFailed("Invalid font".to_string()))?;
212
213        let outlines = font_ref.outline_glyphs();
214        let glyph_id_obj = skrifa::GlyphId::new(glyph_id);
215
216        // Try to get the outline - bitmap glyphs won't have one
217        let glyph = match outlines.get(glyph_id_obj) {
218            Some(g) => g,
219            None => {
220                let bitmap_strikes = font_ref.bitmap_strikes();
221                if !bitmap_strikes.is_empty() {
222                    // Bitmap-only glyph - SVG paths can't represent bitmap data directly.
223                    return Ok(GlyphOutline::BitmapOnly);
224                }
225                // No bitmap data either - this glyph truly doesn't exist
226                return Err(ExportError::EncodingFailed(format!(
227                    "Glyph {} not found (no outline or bitmap data)",
228                    glyph_id
229                ))
230                .into());
231            },
232        };
233
234        // Extract at units_per_em size, then apply scale in the path builder
235        // This avoids double-scaling issues
236        let mut path_builder = SvgPathBuilder::new(scale);
237
238        let size = skrifa::instance::Size::new(font.units_per_em() as f32);
239        let location = skrifa::instance::LocationRef::default();
240        let settings = skrifa::outline::DrawSettings::unhinted(size, location);
241
242        glyph
243            .draw(settings, &mut path_builder)
244            .map_err(|_| ExportError::EncodingFailed("Outline extraction failed".to_string()))?;
245
246        let path = path_builder
247            .finish()
248            .map_err(|e| ExportError::WriteFailed(e.to_string()))?;
249        if path.is_empty() {
250            return Ok(GlyphOutline::EmptyOutline);
251        }
252        Ok(GlyphOutline::Path(path))
253    }

255    fn write_missing_glyph_placeholder(
256        &self,
257        svg: &mut String,
258        x: f32,
259        y: f32,
260        advance: f32,
261        em_height: f32,
262        foreground: Color,
263    ) -> Result<()> {
264        let height = em_height.max(1.0);
265        let width = advance.max(height * 0.5).max(1.0);
266        let y_top = y - height;
267
268        writeln!(
269            svg,
270            r#"  <rect class="typf-missing-glyph" x="{:.2}" y="{:.2}" width="{:.2}" height="{:.2}" fill="none" stroke="rgb({},{},{})" stroke-opacity="{:.2}" stroke-width="1"/>"#,
271            x,
272            y_top,
273            width,
274            height,
275            foreground.r,
276            foreground.g,
277            foreground.b,
278            (foreground.a as f32 / 255.0) * 0.35,
279        )
280        .map_err(|e| ExportError::WriteFailed(e.to_string()))?;
281
282        Ok(())
283    }

285    /// Try to render a bitmap glyph and embed it as a base64 PNG image element
286    ///
287    /// Returns `Ok(Some(image_element))` if successful, `Ok(None)` if no bitmap available.
288    #[cfg(feature = "bitmap-embed")]
289    fn try_embed_bitmap_glyph(
290        &self,
291        font: &Arc<dyn FontRef>,
292        glyph_id: u32,
293        font_size: f32,
294        x: f32,
295        y: f32,
296    ) -> Result<Option<String>> {
297        use typf_render_color::bitmap::{has_bitmap_glyphs, render_bitmap_glyph};
298
299        let font_data = font.data();
300
301        // Check if font has bitmap glyphs
302        if !has_bitmap_glyphs(font_data) {
303            return Ok(None);
304        }
305
306        // Try to render the bitmap glyph
307        let pixmap = match render_bitmap_glyph(font_data, glyph_id, font_size) {
308            Ok(p) => p,
309            Err(_) => return Ok(None), // No bitmap for this glyph
310        };
311
312        // Encode to PNG
313        let png_data = pixmap
314            .encode_png()
315            .map_err(|e| ExportError::EncodingFailed(format!("PNG encoding failed: {}", e)))?;
316
317        // Base64 encode
318        let base64_data = BASE64_STANDARD.encode(&png_data);
319
320        // Create SVG image element
321        // Position: x is horizontal position, y needs adjustment for baseline
322        // The image should be positioned so its bottom aligns with the baseline
323        let img_width = pixmap.width() as f32;
324        let img_height = pixmap.height() as f32;
325
326        // Adjust y position: move up by image height to align bottom with baseline
327        let img_y = y - img_height;
328
329        let image_element = format!(
330            r#"  <image x="{:.2}" y="{:.2}" width="{}" height="{}" href="data:image/png;base64,{}"/>"#,
331            x, img_y, img_width, img_height, base64_data
332        );
333
334        Ok(Some(image_element))
335    }

351/// SVG path builder implementing skrifa's OutlinePen
352struct SvgPathBuilder {
353    commands: String,
354    scale: f32,
355    /// First formatting error seen while building the path, if any.
356    ///
357    /// `OutlinePen` methods cannot return a `Result`, so any write failure is
358    /// captured here and surfaced by [`SvgPathBuilder::finish`] instead of being
359    /// silently discarded.
360    write_error: Option<std::fmt::Error>,
361}
362
363impl SvgPathBuilder {
364    fn new(scale: f32) -> Self {
365        Self {
366            commands: String::new(),
367            scale,
368            write_error: None,
369        }
370    }
371
372    /// Record the outcome of a `write!` call, keeping the first error seen.
373    fn record(&mut self, result: std::fmt::Result) {
374        if self.write_error.is_none() {
375            if let Err(e) = result {
376                self.write_error = Some(e);
377            }
378        }
379    }
380
381    /// Finish building, returning the path string or the first write error.
382    fn finish(self) -> std::result::Result<String, std::fmt::Error> {
383        match self.write_error {
384            Some(e) => Err(e),
385            None => Ok(self.commands),
386        }
387    }
388}
389
390impl skrifa::outline::OutlinePen for SvgPathBuilder {
391    fn move_to(&mut self, x: f32, y: f32) {
392        let x = x * self.scale;
393        let y = -y * self.scale; // Flip Y for SVG coordinate system
394        let r = write!(&mut self.commands, "M{:.2},{:.2}", x, y);
395        self.record(r);
396    }
397
398    fn line_to(&mut self, x: f32, y: f32) {
399        let x = x * self.scale;
400        let y = -y * self.scale;
401        let r = write!(&mut self.commands, "L{:.2},{:.2}", x, y);
402        self.record(r);
403    }
404
405    fn quad_to(&mut self, cx: f32, cy: f32, x: f32, y: f32) {
406        let cx = cx * self.scale;
407        let cy = -cy * self.scale;
408        let x = x * self.scale;
409        let y = -y * self.scale;
410        let r = write!(&mut self.commands, "Q{:.2},{:.2} {:.2},{:.2}", cx, cy, x, y);
411        self.record(r);
412    }
413
414    fn curve_to(&mut self, cx0: f32, cy0: f32, cx1: f32, cy1: f32, x: f32, y: f32) {
415        let cx0 = cx0 * self.scale;
416        let cy0 = -cy0 * self.scale;
417        let cx1 = cx1 * self.scale;
418        let cy1 = -cy1 * self.scale;
419        let x = x * self.scale;
420        let y = -y * self.scale;
421        let r = write!(
422            &mut self.commands,
423            "C{:.2},{:.2} {:.2},{:.2} {:.2},{:.2}",
424            cx0, cy0, cx1, cy1, x, y
425        );
426        self.record(r);
427    }
428
429    fn close(&mut self) {
430        self.commands.push('Z');
431    }
432}
