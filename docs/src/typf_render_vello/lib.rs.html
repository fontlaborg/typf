1//! Vello GPU renderer backend for typf.
2//!
3//! This crate provides GPU-accelerated text rendering using the Vello hybrid renderer.
4//! It uses wgpu for GPU access and provides high-performance rendering by leveraging
5//! GPU compute for path rasterization.
6//!
7//! # Example
8//!
9//! ```ignore
10//! use typf_render_vello::VelloRenderer;
11//! use typf_core::traits::Renderer;
12//!
13//! let renderer = VelloRenderer::new()?;
14//! let output = renderer.render(&shaped_result, font, &params)?;
15//! ```

36/// Errors specific to the Vello GPU renderer.
37#[derive(Error, Debug)]
38pub enum VelloError {
39    /// Failed to initialize wgpu adapter.
40    #[error("Failed to find a suitable GPU adapter")]
41    NoAdapter,
42    /// Failed to create wgpu device.
43    #[error("Failed to create GPU device: {0}")]
44    DeviceCreation(String),
45    /// Rendering failed.
46    #[error("Rendering failed: {0}")]
47    RenderFailed(String),
48    /// GPU poll/readback failed.
49    #[error("GPU readback failed: {0}")]
50    ReadbackFailed(String),
51}
52
53impl From<VelloError> for RenderError {
54    fn from(e: VelloError) -> Self {
55        RenderError::BackendError(e.to_string())
56    }
57}
58
59impl From<VelloError> for typf_core::TypfError {
60    fn from(e: VelloError) -> Self {
61        typf_core::TypfError::RenderingFailed(RenderError::BackendError(e.to_string()))
62    }
63}

65/// Configuration for the Vello GPU renderer.
66#[derive(Debug, Clone)]
67pub struct VelloConfig {
68    /// Whether to use CPU shaders as fallback (slower but more compatible).
69    pub use_cpu_fallback: bool,
70    /// Power preference for GPU adapter selection.
71    pub power_preference: wgpu::PowerPreference,
72}

74impl Default for VelloConfig {
75    fn default() -> Self {
76        Self {
77            use_cpu_fallback: false,
78            power_preference: wgpu::PowerPreference::HighPerformance,
79        }
80    }

83/// GPU context holding wgpu device and queue.
84struct GpuContext {
85    device: Device,
86    queue: Queue,
87}
88
89impl GpuContext {
90    fn new(config: &VelloConfig) -> Result<Self, VelloError> {
91        pollster::block_on(Self::new_async(config))
92    }
93
94    async fn new_async(config: &VelloConfig) -> Result<Self, VelloError> {
95        let instance = wgpu::Instance::default();
96
97        let adapter = instance
98            .request_adapter(&wgpu::RequestAdapterOptions {
99                power_preference: config.power_preference,
100                force_fallback_adapter: config.use_cpu_fallback,
101                compatible_surface: None,
102            })
103            .await
104            .map_err(|_| VelloError::NoAdapter)?;
105
106        let (device, queue) = adapter
107            .request_device(&wgpu::DeviceDescriptor {
108                label: Some("typf-vello"),
109                required_features: wgpu::Features::empty(),
110                ..Default::default()
111            })
112            .await
113            .map_err(|e| VelloError::DeviceCreation(e.to_string()))?;
114
115        Ok(Self { device, queue })
116    }
117}

119/// Vello GPU renderer for high-performance text rendering.
120///
121/// This renderer uses the Vello hybrid CPU/GPU rendering engine with wgpu
122/// for GPU acceleration. It provides excellent performance for rendering
123/// large amounts of text or text at large sizes.
124pub struct VelloRenderer {
125    config: VelloConfig,
126    gpu: GpuContext,
127}

138impl VelloRenderer {
139    /// Creates a new Vello GPU renderer with default configuration.
140    ///
141    /// # Errors
142    ///
143    /// Returns an error if GPU initialization fails (no suitable adapter or device creation fails).
144    pub fn new() -> Result<Self, VelloError> {
145        Self::with_config(VelloConfig::default())
146    }
147
148    /// Creates a new Vello GPU renderer with custom configuration.
149    ///
150    /// # Errors
151    ///
152    /// Returns an error if GPU initialization fails.
153    pub fn with_config(config: VelloConfig) -> Result<Self, VelloError> {
154        let gpu = GpuContext::new(&config)?;
155        Ok(Self { config, gpu })
156    }
157
158    /// Convert typf color to Vello AlphaColor.
159    fn to_vello_color(color: Color) -> AlphaColor<vello_common::peniko::color::Srgb> {
160        AlphaColor::from_rgba8(color.r, color.g, color.b, color.a)
161    }
162
163    /// Convert typf ShapingResult glyphs to Vello Glyph format.
164    ///
165    /// Only the glyph id and pen position are carried over. Color glyph tables
166    /// are **not** consulted here: `COLR` v1 (layered/gradient), the `SVG` table,
167    /// and bitmap strikes (`sbix`/`CBDT`/`EBDT`) are unsupported by this backend
168    /// and render as plain outlines filled with the single foreground paint.
169    fn convert_glyphs(shaped: &ShapingResult) -> Vec<Glyph> {
170        shaped
171            .glyphs
172            .iter()
173            .map(|g| Glyph {
174                id: g.id,
175                x: g.x,
176                y: g.y,
177            })
178            .collect()
179    }
180
181    /// Render to a GPU texture and read back to CPU.
182    fn render_to_bitmap(
183        &self,
184        scene: &Scene,
185        width: u32,
186        height: u32,
187    ) -> Result<Vec<u8>, VelloError> {
188        let device = &self.gpu.device;
189        let queue = &self.gpu.queue;
190
191        // Create render target texture
192        let texture = device.create_texture(&wgpu::TextureDescriptor {
193            label: Some("typf-vello-target"),
194            size: wgpu::Extent3d {
195                width,
196                height,
197                depth_or_array_layers: 1,
198            },
199            mip_level_count: 1,
200            sample_count: 1,
201            dimension: wgpu::TextureDimension::D2,
202            format: wgpu::TextureFormat::Rgba8Unorm,
203            usage: wgpu::TextureUsages::RENDER_ATTACHMENT | wgpu::TextureUsages::COPY_SRC,
204            view_formats: &[],
205        });
206        let texture_view = texture.create_view(&wgpu::TextureViewDescriptor::default());
207
208        // Create Vello renderer
209        let mut renderer = vello_hybrid::Renderer::new(
210            device,
211            &RenderTargetConfig {
212                format: wgpu::TextureFormat::Rgba8Unorm,
213                width,
214                height,
215            },
216        );
217
218        let render_size = RenderSize { width, height };
219
220        // Create command encoder and render
221        let mut encoder = device.create_command_encoder(&wgpu::CommandEncoderDescriptor {
222            label: Some("typf-vello-render"),
223        });
224
225        renderer
226            .render(
227                scene,
228                device,
229                queue,
230                &mut encoder,
231                &render_size,
232                &texture_view,
233            )
234            .map_err(|e| VelloError::RenderFailed(format!("{:?}", e)))?;
235
236        // Create buffer for readback (with row padding to 256 bytes)
237        let bytes_per_row = (width * 4).next_multiple_of(256);
238        let buffer_size = (bytes_per_row as u64) * (height as u64);
239        let readback_buffer = device.create_buffer(&wgpu::BufferDescriptor {
240            label: Some("typf-vello-readback"),
241            size: buffer_size,
242            usage: wgpu::BufferUsages::COPY_DST | wgpu::BufferUsages::MAP_READ,
243            mapped_at_creation: false,
244        });
245
246        // Copy texture to buffer
247        encoder.copy_texture_to_buffer(
248            wgpu::TexelCopyTextureInfo {
249                texture: &texture,
250                mip_level: 0,
251                origin: wgpu::Origin3d::ZERO,
252                aspect: wgpu::TextureAspect::All,
253            },
254            wgpu::TexelCopyBufferInfo {
255                buffer: &readback_buffer,
256                layout: wgpu::TexelCopyBufferLayout {
257                    offset: 0,
258                    bytes_per_row: Some(bytes_per_row),
259                    rows_per_image: None,
260                },
261            },
262            wgpu::Extent3d {
263                width,
264                height,
265                depth_or_array_layers: 1,
266            },
267        );
268
269        // Submit and wait
270        queue.submit([encoder.finish()]);
271
272        // Map buffer for reading
273        let buffer_slice = readback_buffer.slice(..);
274        buffer_slice.map_async(wgpu::MapMode::Read, |_| {});
275        device
276            .poll(wgpu::PollType::wait())
277            .map_err(|e| VelloError::ReadbackFailed(format!("{:?}", e)))?;
278
279        // Read data and remove row padding
280        let mapped = buffer_slice.get_mapped_range();
281        let mut result = Vec::with_capacity((width * height * 4) as usize);
282        for row in mapped.chunks_exact(bytes_per_row as usize) {
283            result.extend_from_slice(&row[0..(width * 4) as usize]);
284        }
285        drop(mapped);
286        readback_buffer.unmap();
287
288        Ok(result)
289    }

292impl Renderer for VelloRenderer {
293    fn name(&self) -> &'static str {
294        "vello"
295    }
296
297    /// Render shaped glyphs to an RGBA bitmap on the GPU.
298    ///
299    /// Unsupported glyph formats: this backend rasterizes glyph **outlines**
300    /// only, filled with a single solid foreground paint. `COLR` v1
301    /// (layered/gradient color), `SVG`-table glyphs, and bitmap strikes
302    /// (`sbix`/`CBDT`/`EBDT`) are not handled — such glyphs appear as plain
303    /// monochrome outlines (or blank, if they have no outline). Use the
304    /// `skia`/`zeno` backends for full color-glyph support.
305    fn render(
306        &self,
307        shaped: &ShapingResult,
308        font: Arc<dyn FontRef>,
309        params: &RenderParams,
310    ) -> Result<RenderOutput> {
311        log::debug!("VelloRenderer: Rendering {} glyphs", shaped.glyphs.len());
312
313        let padding = params.padding as f32;
314        let font_size = shaped.advance_height;
315
316        // Baseline standardization (metrics-first):
317        // - Prefer font ascent/descent (stable across strings)
318        // - Fall back to a heuristic if font metrics are unavailable.
319        let (ascent, descent) = font
320            .metrics()
321            .filter(|m| m.units_per_em > 0 && (m.ascent != 0 || m.descent != 0))
322            .map(|m| {
323                let scale = font_size / (m.units_per_em as f32);
324                let ascent = (m.ascent as f32).max(0.0) * scale;
325                let descent = (m.descent as f32).abs() * scale;
326                (ascent, descent)
327            })
328            .unwrap_or((font_size * 0.8, font_size * 0.2));
329
330        // Calculate canvas dimensions
331        let width = (shaped.advance_width + padding * 2.0).ceil() as u32;
332        let height = (ascent + descent + padding * 2.0).ceil() as u32;
333
334        // Sanity check dimensions
335        if width == 0 || height == 0 {
336            return Err(RenderError::ZeroDimensions { width, height }.into());
337        }
338
339        // vello_hybrid uses u16 for dimensions internally
340        let width_u16 = width.min(u16::MAX as u32) as u16;
341        let height_u16 = height.min(u16::MAX as u32) as u16;
342
343        // Create scene
344        let mut scene = Scene::new(width_u16, height_u16);
345
346        // Set background color if specified
347        if let Some(bg) = params.background {
348            scene.set_paint(Self::to_vello_color(bg));
349            scene.fill_rect(&vello_common::kurbo::Rect::new(
350                0.0,
351                0.0,
352                width as f64,
353                height as f64,
354            ));
355        }
356
357        // Set foreground color
358        scene.set_paint(Self::to_vello_color(params.foreground));
359
360        // Create font data for Vello. Prefer shared bytes to avoid a per-render copy.
361        let font_blob = if let Some(shared) = font.data_shared() {
362            Blob::new(shared)
363        } else {
364            font.data().to_vec().into()
365        };
366        let font_data = FontData::new(font_blob, 0);
367
368        // Convert glyphs
369        let glyphs = Self::convert_glyphs(shaped);
370
371        if !glyphs.is_empty() {
372            // Baseline position: padding + ascent (top of canvas + ascender space)
373            let baseline_y = padding + ascent;
374
375            // Set transform for positioning
376            scene.set_transform(Affine::translate((padding as f64, baseline_y as f64)));
377
378            // Render glyphs using glyph_run builder
379            scene
380                .glyph_run(&font_data)
381                .font_size(font_size)
382                .fill_glyphs(glyphs.into_iter());
383        }
384
385        // Render scene to bitmap
386        let rgba_data = self.render_to_bitmap(&scene, width, height)?;
387
388        Ok(RenderOutput::Bitmap(BitmapData {
389            width,
390            height,
391            format: BitmapFormat::Rgba8,
392            data: rgba_data,
393        }))
394    }
395
396    fn supports_format(&self, format: &str) -> bool {
397        matches!(format, "bitmap" | "rgba" | "rgb")
398    }
399
400    fn clear_cache(&self) {
401        // GPU resources are recreated per render, no persistent cache
402    }
403}

419    #[test]
420    fn test_color_conversion() {
421        let color = Color::rgba(255, 128, 64, 200);
422        let vello_color = VelloRenderer::to_vello_color(color);
423        // AlphaColor stores components as f32 in [0, 1] range
424        assert!((vello_color.components[0] - 1.0).abs() < 0.01); // R
425        assert!((vello_color.components[1] - 0.5).abs() < 0.05); // G
426        assert!((vello_color.components[2] - 0.25).abs() < 0.05); // B
427        assert!((vello_color.components[3] - 0.78).abs() < 0.05); // A
428    }
