1//! Divide and conquer: multi-core rendering made simple
2//!
3//! Why render one glyph at a time when you have multiple CPU cores? This module
4//! splits your text across available threads, letting each one work on a portion
5//! of the glyphs. Rayon handles the thread management—we just reap the speed
6//! benefits.
7
8use rayon::prelude::*;
9use typf_core::{
10    types::{PositionedGlyph, ShapingResult},
11    Color,
12};

14/// Your multi-core orchestrator: coordinating glyph rendering across threads
15pub struct ParallelRenderer {
16    /// How many cores to enlist (0 = let Rayon decide optimally)
17    thread_count: usize,
18}

20impl ParallelRenderer {
21    /// Ready your parallel rendering team
22    pub fn new() -> Self {
23        Self {
24            thread_count: 0, // Let Rayon's thread pool choose wisely
25        }
26    }

29    pub fn with_threads(thread_count: usize) -> Self {

33    /// Parallel glyph rendering: faster when you have many cores
34    ///
35    /// We split your glyph list into chunks, hand each to a different thread,
36    /// then carefully blend results back together. Perfect for large paragraphs
37    /// or batch processing where speed matters.
38    pub fn render_parallel(
39        &self,
40        glyphs: &[PositionedGlyph],
41        canvas_width: u32,
42        canvas_height: u32,
43        glyph_renderer: impl Fn(&PositionedGlyph) -> Vec<u8> + Send + Sync,
44        color: Color,
45        background: Option<Color>,
46    ) -> Vec<u8> {
47        // Set up our rendering canvas with proper dimensions
48        let canvas_size = (canvas_width * canvas_height * 4) as usize;
49        let mut canvas = vec![0u8; canvas_size];
50
51        // Apply background color before任何 glyph rendering begins
52        if let Some(bg) = background {
53            for pixel in canvas.chunks_exact_mut(4) {
54                pixel[0] = bg.r;
55                pixel[1] = bg.g;
56                pixel[2] = bg.b;
57                pixel[3] = bg.a;
58            }
59        }
60
61        // Override Rayon's default thread pool if you know better
62        if self.thread_count > 0 {
63            rayon::ThreadPoolBuilder::new()
64                .num_threads(self.thread_count)
65                .build()
66                .ok();
67        }
68
69        // Send each glyph to a thread and collect the results
70        let glyph_bitmaps: Vec<_> = glyphs
71            .par_iter()
72            .map(|glyph| {
73                let bitmap = glyph_renderer(glyph);
74                (glyph, bitmap)
75            })
76            .collect();
77
78        // Carefully blend results back together (order matters for alpha)
79        for (glyph, bitmap) in glyph_bitmaps {
80            self.composite_glyph(
81                &mut canvas,
82                canvas_width,
83                &bitmap,
84                glyph.x as i32,
85                glyph.y as i32,
86                color,
87            );
88        }
89
90        canvas
91    }

93    /// Regional parallelism: divide the canvas, conquer the text
94    ///
95    /// Instead of splitting glyphs, we split the canvas itself. Each thread
96    /// works on a different horizontal region. This shines when you have
97    /// tall text blocks with minimal cross-line glyph overlap.
98    pub fn render_regions(
99        &self,
100        shaped: &ShapingResult,
101        canvas_width: u32,
102        canvas_height: u32,
103        glyph_renderer: impl Fn(&PositionedGlyph) -> Vec<u8> + Send + Sync,
104        color: Color,
105    ) -> Vec<u8> {
106        // Split canvas into sensible horizontal strips
107        let region_height = 64; // Typical line height
108        let num_regions = (canvas_height / region_height).max(1);
109
110        // Assign each glyph to its home region based on Y position
111        let mut regions = vec![Vec::new(); num_regions as usize];
112        for glyph in &shaped.glyphs {
113            let region_idx = (glyph.y as u32 / region_height).min(num_regions - 1) as usize;
114            regions[region_idx].push(glyph.clone());
115        }
116
117        // Let each thread paint its portion of the canvas independently
118        let rendered_regions: Vec<_> = regions
119            .par_iter()
120            .enumerate()
121            .map(|(idx, glyphs)| {
122                let region_y = idx as u32 * region_height;
123                let mut region_canvas = vec![0u8; (canvas_width * region_height * 4) as usize];
124
125                // Render glyphs in this region
126                for glyph in glyphs {
127                    let bitmap = glyph_renderer(glyph);
128                    let local_y = glyph.y - region_y as f32;
129                    self.composite_glyph(
130                        &mut region_canvas,
131                        canvas_width,
132                        &bitmap,
133                        glyph.x as i32,
134                        local_y as i32,
135                        color,
136                    );
137                }
138
139                (idx, region_canvas)
140            })
141            .collect();
142
143        // Stitch the regions together into our final masterpiece
144        let mut canvas = vec![0u8; (canvas_width * canvas_height * 4) as usize];
145        for (idx, region_data) in rendered_regions {
146            let region_y = idx as u32 * region_height;
147            let start_idx = (region_y * canvas_width * 4) as usize;
148            let region_size = region_data.len().min(canvas.len() - start_idx);
149            canvas[start_idx..start_idx + region_size].copy_from_slice(&region_data[..region_size]);
150        }
151
152        canvas
153    }
154
155    /// The final touch: blending individual glyphs into the canvas
156    fn composite_glyph(
157        &self,
158        canvas: &mut [u8],
159        canvas_width: u32,
160        glyph_bitmap: &[u8],
161        x: i32,
162        y: i32,
163        color: Color,
164    ) {
165        // Alpha blending with proper color mixing (SIMD would be faster here)
166        let glyph_size = (glyph_bitmap.len() as f32).sqrt() as u32;
167        let canvas_height = canvas.len() as u32 / (canvas_width * 4);
168
169        for gy in 0..glyph_size {
170            for gx in 0..glyph_size {
171                let px = x + gx as i32;
172                let py = y + gy as i32;
173
174                if px < 0 || py < 0 || px >= canvas_width as i32 || py >= canvas_height as i32 {
175                    continue;
176                }
177
178                let coverage = glyph_bitmap[(gy * glyph_size + gx) as usize];
179                if coverage == 0 {
180                    continue;
181                }
182
183                let canvas_idx = ((py as u32 * canvas_width + px as u32) * 4) as usize;
184
185                // Classic Porter-Duff blending for smooth edges
186                let alpha = (coverage as f32 / 255.0) * (color.a as f32 / 255.0);
187                let inv_alpha = 1.0 - alpha;
188
189                canvas[canvas_idx] =
190                    (canvas[canvas_idx] as f32 * inv_alpha + color.r as f32 * alpha) as u8;
191                canvas[canvas_idx + 1] =
192                    (canvas[canvas_idx + 1] as f32 * inv_alpha + color.g as f32 * alpha) as u8;
193                canvas[canvas_idx + 2] =
194                    (canvas[canvas_idx + 2] as f32 * inv_alpha + color.b as f32 * alpha) as u8;
195                canvas[canvas_idx + 3] =
196                    ((canvas[canvas_idx + 3] as f32 * inv_alpha + 255.0 * alpha).min(255.0)) as u8;
197            }
198        }
199    }
200}

208/// Your performance dashboard: measuring parallel benefits
209#[derive(Debug, Clone, Default)]
210pub struct ParallelStats {
211    pub total_glyphs: usize,
212    pub threads_used: usize,
213    pub render_time_ms: u128,
214    pub composite_time_ms: u128,
215}
216
217impl ParallelStats {
218    /// Calculate speedup vs single-threaded
219    pub fn speedup(&self) -> f32 {
220        if self.threads_used <= 1 {
221            1.0
222        } else {
223            let total_time = self.render_time_ms + self.composite_time_ms;
224            let estimated_sequential =
225                self.total_glyphs as u128 * total_time / self.threads_used as u128;
226            estimated_sequential as f32 / total_time as f32
227        }
228    }
229}

236    #[test]
237    fn test_parallel_renderer() {
238        let renderer = ParallelRenderer::new();
239
240        let glyphs = vec![
241            PositionedGlyph {
242                id: 42,
243                x: 0.0,
244                y: 0.0,
245                advance: 10.0,
246                cluster: 0,
247            },
248            PositionedGlyph {
249                id: 43,
250                x: 10.0,
251                y: 0.0,
252                advance: 10.0,
253                cluster: 1,
254            },
255        ];
256
257        // Mock glyph renderer
258        let glyph_renderer = |_glyph: &PositionedGlyph| {
259            vec![255u8; 100] // 10x10 glyph
260        };
261
262        let canvas = renderer.render_parallel(
263            &glyphs,
264            100,
265            20,
266            glyph_renderer,
267            Color::rgba(0, 0, 0, 255),
268            None,
269        );
270
271        assert_eq!(canvas.len(), 100 * 20 * 4);
272    }
