1//! Glyph bitmap cache for efficient repeated rendering
2//!
3//! When rendering text, the same glyph often appears multiple times. This cache
4//! stores rendered glyph bitmaps keyed by (font, glyph_id, size, variations) to
5//! avoid redundant rasterization work.
6
7use std::collections::hash_map::DefaultHasher;
8use std::collections::HashMap;
9use std::hash::{Hash, Hasher};
10use std::sync::RwLock;
11
12use crate::rasterizer::GlyphBitmap;
13
14/// Cache key for rendered glyphs
15///
16/// Uniquely identifies a rendered glyph by its font, glyph ID, size, and variations.
17#[derive(Debug, Clone, PartialEq, Eq, Hash)]
18pub struct GlyphCacheKey {
19    /// Hash of font data (identifies the font)
20    pub font_id: u64,
21    /// Glyph ID within the font
22    pub glyph_id: u32,
23    /// Size in fixed-point (size * 100 for hash stability)
24    pub size: u32,
25    /// Hinting mode (0 = none, 1 = TrueType) and hint ppem (0 = from size),
26    /// so hinted and unhinted bitmaps never alias.
27    pub hinting: u8,
28    pub ppem: u32,
29    /// Hash of variation coordinates
30    pub variations_hash: u64,
31}
32
33impl GlyphCacheKey {
34    /// Tag the key with the hinting configuration.
35    pub fn with_hinting(mut self, hinting: typf_core::HintingMode, ppem: Option<u32>) -> Self {
36        self.hinting = match hinting {
37            typf_core::HintingMode::None => 0,
38            typf_core::HintingMode::TrueType => 1,
39            _ => 255,
40        };
41        self.ppem = ppem.unwrap_or(0);
42        self
43    }
44
45    /// Create a new cache key
46    pub fn new(font_data: &[u8], glyph_id: u32, size: f32, variations: &[(String, f32)]) -> Self {
47        // Hash font data
48        let mut hasher = DefaultHasher::new();
49        font_data.hash(&mut hasher);
50        let font_id = hasher.finish();
51
52        // Hash variations
53        let mut var_hasher = DefaultHasher::new();
54        for (tag, val) in variations {
55            tag.hash(&mut var_hasher);
56            ((val * 1000.0) as i32).hash(&mut var_hasher);
57        }
58        let variations_hash = var_hasher.finish();
59
60        Self {
61            font_id,
62            glyph_id,
63            size: (size * 100.0) as u32,
64            hinting: 0,
65            ppem: 0,
66            variations_hash,
67        }
68    }
69}

71/// LRU-style glyph cache with configurable capacity
72pub struct GlyphCache {
73    cache: RwLock<HashMap<GlyphCacheKey, GlyphBitmap>>,
74    capacity: usize,
75    hits: RwLock<u64>,
76    misses: RwLock<u64>,
77}
78
79impl GlyphCache {
80    /// Create a new glyph cache with specified capacity
81    pub fn new(capacity: usize) -> Self {
82        Self {
83            cache: RwLock::new(HashMap::with_capacity(capacity)),
84            capacity,
85            hits: RwLock::new(0),
86            misses: RwLock::new(0),
87        }
88    }
89
90    /// Get a cached glyph bitmap if available
91    pub fn get(&self, key: &GlyphCacheKey) -> Option<GlyphBitmap> {
92        let cache = self.cache.read().ok()?;
93        if let Some(bitmap) = cache.get(key) {
94            if let Ok(mut hits) = self.hits.write() {
95                *hits += 1;
96            }
97            Some(bitmap.clone())
98        } else {
99            if let Ok(mut misses) = self.misses.write() {
100                *misses += 1;
101            }
102            None
103        }
104    }
105
106    /// Insert a glyph bitmap into the cache
107    pub fn insert(&self, key: GlyphCacheKey, bitmap: GlyphBitmap) {
108        let mut cache = match self.cache.write() {
109            Ok(c) => c,
110            Err(_) => return,
111        };
112
113        // Simple eviction: clear half the cache when full
114        if cache.len() >= self.capacity {
115            let keys_to_remove: Vec<_> = cache.keys().take(self.capacity / 2).cloned().collect();
116            for k in keys_to_remove {
117                cache.remove(&k);
118            }
119        }
120
121        cache.insert(key, bitmap);
122    }
123
124    /// Get cache hit rate (0.0 to 1.0)
125    pub fn hit_rate(&self) -> f64 {
126        let hits = self.hits.read().map(|h| *h).unwrap_or(0);
127        let misses = self.misses.read().map(|m| *m).unwrap_or(0);
128        let total = hits + misses;
129        if total == 0 {
130            0.0
131        } else {
132            hits as f64 / total as f64
133        }
134    }
135
136    /// Get cache statistics
137    pub fn stats(&self) -> GlyphCacheStats {
138        let cache = self.cache.read().ok();
139        GlyphCacheStats {
140            size: cache.map(|c| c.len()).unwrap_or(0),
141            capacity: self.capacity,
142            hits: self.hits.read().map(|h| *h).unwrap_or(0),
143            misses: self.misses.read().map(|m| *m).unwrap_or(0),
144        }
145    }
146
147    /// Clear the cache
148    pub fn clear(&self) {
149        if let Ok(mut cache) = self.cache.write() {
150            cache.clear();
151        }
152    }
153}

155/// Glyph cache statistics
156#[derive(Debug, Clone)]
157pub struct GlyphCacheStats {
158    pub size: usize,
159    pub capacity: usize,
160    pub hits: u64,
161    pub misses: u64,
162}
163
164impl GlyphCacheStats {
165    pub fn hit_rate(&self) -> f64 {
166        let total = self.hits + self.misses;
167        if total == 0 {
168            0.0
169        } else {
170            self.hits as f64 / total as f64
171        }
172    }
173}

175#[cfg(test)]

179    #[test]
180    fn test_cache_key_hinting_does_not_alias() {
181        use typf_core::HintingMode;
182        let base = GlyphCacheKey::new(b"font1", 65, 16.0, &[]);
183        let none = base.clone().with_hinting(HintingMode::None, None);
184        let tt16 = base.clone().with_hinting(HintingMode::TrueType, Some(16));
185        let tt17 = base.clone().with_hinting(HintingMode::TrueType, Some(17));
186        assert_eq!(base, none, "None hinting without ppem equals the plain key");
187        assert_ne!(none, tt16, "hinting mode must be part of the key");
188        assert_ne!(tt16, tt17, "ppem must be part of the key");
189    }
190
191    #[test]
192    fn test_cache_key_creation() {
193        let key1 = GlyphCacheKey::new(b"font1", 65, 16.0, &[]);
194        let key2 = GlyphCacheKey::new(b"font1", 65, 16.0, &[]);
195        let key3 = GlyphCacheKey::new(b"font2", 65, 16.0, &[]);
196
197        assert_eq!(key1, key2);
198        assert_ne!(key1, key3);
199    }

201    #[test]
202    fn test_variations_affect_key() {
203        let key1 = GlyphCacheKey::new(b"font", 65, 16.0, &[("wght".to_string(), 400.0)]);
204        let key2 = GlyphCacheKey::new(b"font", 65, 16.0, &[("wght".to_string(), 700.0)]);
205        let key3 = GlyphCacheKey::new(b"font", 65, 16.0, &[]);
206
207        assert_ne!(key1, key2, "Different weights should have different keys");
208        assert_ne!(key1, key3, "With/without variations should differ");
209    }

213        let cache = GlyphCache::new(100);

218            left: 1,
219            top: 10,
