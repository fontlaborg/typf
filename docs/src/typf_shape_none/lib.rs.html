1//! When you just need characters laid out: the simplest possible shaper
2//!
3//! This is our "just the basics" shaper. No ligatures, no kerning, no complex
4//! script support. Just takes your text, finds the matching glyphs, and lays
5//! them out left-to-right. Perfect for ASCII, debugging, or when you don't
6//! want the complexity of HarfBuzz.
7
8use std::sync::Arc;
9use typf_core::{
10    error::Result,
11    traits::{FontRef, Shaper},
12    types::{PositionedGlyph, ShapingResult},
13    ShapingParams,
14};
15
16/// The simplest shaper: one character = one glyph, laid out left-to-right
17pub struct NoneShaper;
18
19impl NoneShaper {
20    /// Creates a new shaper that does the absolute minimum
21    pub fn new() -> Self {
22        Self
23    }
24}
25
26impl Default for NoneShaper {
27    fn default() -> Self {
28        Self::new()
29    }
30}
31
32impl Shaper for NoneShaper {
33    fn name(&self) -> &'static str {
34        "none"
35    }
36
37    fn shape(
38        &self,
39        text: &str,
40        font: Arc<dyn FontRef>,
41        params: &ShapingParams,
42    ) -> Result<ShapingResult> {
43        log::debug!("NoneShaper: Shaping {} chars", text.chars().count());
44
45        let mut glyphs = Vec::new();
46        let mut x_advance = 0.0;
47        let scale = params.size / font.units_per_em() as f32;
48
49        // One character becomes one glyph, positioned sequentially
50        for (cluster, ch) in text.char_indices() {
51            // Find which glyph draws this character
52            let glyph_id = font.glyph_id(ch).unwrap_or(0); // Use .notdef (0) if not found
53
54            // Get the glyph's width and scale it to our display size
55            let advance_unscaled = font.advance_width(glyph_id);
56            let advance = advance_unscaled * scale + params.letter_spacing;
57
58            // Place the glyph at the current position
59            glyphs.push(PositionedGlyph {
60                id: glyph_id,
61                x: x_advance,
62                y: 0.0,
63                advance,
64                cluster: cluster as u32,
65            });
66
67            // Move to the right for the next glyph
68            x_advance += advance;
69        }
70
71        Ok(ShapingResult {
72            glyphs,
73            advance_width: x_advance,
74            advance_height: params.size,
75            direction: params.direction,
76        })
77    }
78
79    fn supports_script(&self, _script: &str) -> bool {
80        // We're honest about our limitations
81        false
82    }
83}

85#[cfg(test)]
86mod tests {
87    use super::*;
88    use typf_core::types::Direction;
89
90    // Mock font for testing
91    struct MockFont;
92
93    impl FontRef for MockFont {
94        fn data(&self) -> &[u8] {
95            &[]
96        }
97
98        fn units_per_em(&self) -> u16 {
99            1000
100        }
101
102        fn glyph_id(&self, ch: char) -> Option<u32> {
103            // Simple mapping: ASCII characters to their values
104            if ch.is_ascii() {
105                Some(ch as u32)
106            } else {
107                None
108            }
109        }
110
111        fn advance_width(&self, _glyph_id: u32) -> f32 {
112            500.0 // Fixed advance for simplicity
113        }
114    }
115
116    #[test]
117    fn test_basic_shaping() {
118        let shaper = NoneShaper::new();
119        let font = Arc::new(MockFont);
120        let params = ShapingParams {
121            size: 16.0,
122            ..Default::default()
123        };
124
125        let result = shaper.shape("Hello", font, &params).unwrap();
126
127        assert_eq!(result.glyphs.len(), 5);
128        assert_eq!(result.direction, Direction::LeftToRight);
129
130        // Check that glyphs are positioned sequentially
131        for i in 1..result.glyphs.len() {
132            assert!(result.glyphs[i].x > result.glyphs[i - 1].x);
133        }
134    }
135
136    #[test]
137    fn test_empty_text() {
138        let shaper = NoneShaper::new();
139        let font = Arc::new(MockFont);
140        let params = ShapingParams::default();
141
142        let result = shaper.shape("", font, &params).unwrap();
143
144        assert_eq!(result.glyphs.len(), 0);
145        assert_eq!(result.advance_width, 0.0);
146    }
